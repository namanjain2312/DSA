10.SubsetSum

class Solution {
public:

    int funct(int index,vector<int>&nums,int currSum,vector<vector<int>>&dp)
    {
        if(index==nums.size())
            return dp[index][currSum]=0;

        if(dp[index][currSum]!=-1)
            return dp[index][currSum];

        if(currSum==0)
            return dp[index][currSum]=1;

        int take=-1,notTake=-1;


        if(currSum-nums[index]>=0)
            take=funct(index+1,nums,currSum-nums[index],dp);
        notTake = funct(index+1,nums,currSum,dp);

        if(take==1 || notTake==1)
            return dp[index][currSum]=1;
        else
            return dp[index][currSum]=0;

    }

    bool canPartition(vector<int>& nums)
    {
        // int totalSum=0,currSum=0,index=0;

        // for(int i=0;i<nums.size();i++)
        //     totalSum+=nums[i];
        // if(totalSum%2!=0)
        //     return false;
        // vector<vector<int>>dp(nums.size()+1,vector<int>(totalSum/2+1,-1));
        // return funct(0,nums,totalSum/2,dp);


        int totalSum=0,currSum=0,index=0,take=0,notTake=0;

        for(int i=0;i<nums.size();i++)
            totalSum+=nums[i];
        if(totalSum%2!=0)
            return false;
        totalSum/=2;
        vector<int>dp(totalSum+1,0);
        dp[0]=1;
        for(int i=0;i<nums.size();i++)
        {
            for(int j=totalSum;j>=0;j--)
            {
                if(j-nums[i]>=0)
                {
                    dp[j]=max(dp[j],dp[j-nums[i]]);
                }
            }

        }
        return dp[totalSum];
    }

};
