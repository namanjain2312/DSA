class Solution {
public:
    int lengthOfLIS(vector<int>& nums) {
        int n=nums.size();
        vector<vector<int>>dp(n+1,vector<int>(n+1,0));


        for(int i=n-1;i>=0;i--)
        {
            for(int j=n;j>=0;j--)
            {
                if(j==n || nums[j]<nums[i])
                    dp[i][j]=1+dp[i+1][i];
                dp[i][j]= max(dp[i][j],dp[i+1][j]);
            }
        }
        return dp[0][n];
    }
};
