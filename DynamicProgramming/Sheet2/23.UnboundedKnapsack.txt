23.UnboundedKnapsack

#include<bits/stdc++.h>

int unboundedKnapsack(int n, int w, vector<int> &profit, vector<int> &weight)
{
    vector<vector<int>>dp(n+1,vector<int>(w+1,INT_MIN));

    for(int i=0;i<=n;i++)
        dp[i][0]=0;

    for(int i=1;i<=n;i++)
    {
        int curr=weight[i-1];
        int val=profit[i-1];

        for(int j=1;j<=w;j++)
        {
            int ans=INT_MIN;
            ans=dp[i-1][j];

            int k=1;
            while(true)
            {
                if(j-k*curr>=0)
                {
                    ans=max(ans,dp[i-1][j-k*curr]+val*k);
                }
                else
                    break;
                k++;
            }
            dp[i][j]=ans;
        }
    }
    if(dp[n][w]<0)
        return 0;

    return dp[n][w];

}
