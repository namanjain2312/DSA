1.MaxProductSubarray

class Solution {
public:
    int maxProduct(vector<int>& nums) {
        int n=nums.size();
        long long suffix=1,prefix=1;
        long long out=INT_MIN;

        for(int i=0;i<n;i++)
        {
            if(prefix==0)
                prefix=1;
            if(suffix==0)
                suffix=1;

            prefix=prefix*nums[i];
            suffix=suffix*nums[n-1-i];
            out=max(out,max(prefix,suffix));

        }

        return (int)out;
    }
};
