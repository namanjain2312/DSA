int funct(int index,vector<int>&values,vector<int>&weights,int w,vector<vector<int>>&dp)
{
    if(index==values.size())
        return 0;

    if(dp[index][w]!=-1)
        return dp[index][w];

    int totalValue=0;

    if(w-weights[index]>=0)
        totalValue = values[index] + funct(index+1,values,weights,w-weights[index],dp);

    totalValue=max(totalValue,funct(index+1,values,weights,w,dp));

    return dp[index][w]=totalValue;
}
