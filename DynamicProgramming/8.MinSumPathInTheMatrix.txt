8.MinSumPathInTheMatrix

class Solution {
public:
    int minPath(int row,int col,vector<vector<int>>&dp,vector<vector<int>>& grid)
    {
        int sum;
        if(dp[row][col]!=INT_MAX)
            return dp[row][col];

        if(row==grid.size()-1 && col == grid[0].size()-1)
            sum =  grid[row][col];
        else if(row==grid.size()-1)
            sum = grid[row][col] + minPath(row,col+1,dp,grid);
        else if(col==grid[0].size()-1)
            sum = grid[row][col] + minPath(row+1,col,dp,grid);
        else
        {
            sum=min(minPath(row,col+1,dp,grid),minPath(row+1,col,dp,grid))+grid[row][col];
        }
        return dp[row][col] = sum;

    }



    int minPathSum(vector<vector<int>>& grid) {
        // int n=grid.size(),m=grid[0].size();

        // vector<vector<int>>dp(n,vector<int>(m,INT_MAX));
        // return minPath(0,0,dp,grid);

        int n=grid.size(),m=grid[0].size();
        vector<vector<int>>dp(n,vector<int>(m,INT_MAX));
        for(int i=n-1;i>=0;i--)
        {
            for(int j=m-1;j>=0;j--)
            {
                dp[i][j]=grid[i][j];
                if(i==n-1&&j==m-1)
                    ;
                else if(i==n-1)
                    dp[i][j]+= dp[i][j+1];
                else if(j==m-1)
                    dp[i][j]+=dp[i+1][j];
                else
                    dp[i][j]+= min(dp[i+1][j],dp[i][j+1]);
            }
        }
        return dp[0][0];


    }
};
