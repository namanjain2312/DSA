vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2)
    {
        map<int,int> m ;
        stack<int> s;
        int n = nums2.size(),val;
        vector<int>output;

        for(int i=n-1;i>=0;i--)
        {
            val = nums2[i];
            while(true)
            {
                if(s.empty())
                {
                    m[val]=-1;
                    s.push(val);
                    break;
                }
                if(s.top()<val)
                    s.pop();
                else
                {
                    m[val]=s.top();
                    s.push(val);
                    break;
                }
            }
        }

        for(int i=0;i<nums1.size();i++)
        {
            output.push_back(m[nums1[i]]);
        }

        return output;
}
