#include <bits/stdc++.h>

void insertInStack(stack<int>&s,int currValue)
{
	if(s.empty())
	{
		s.push(currValue);
		return ;
	}
	int topEle=s.top();
	if(topEle<=currValue)
		s.push(currValue);
	else
	{
		s.pop();
		insertInStack(s,currValue);
		s.push(topEle);
	}
	return ;
}

void sortStack(stack<int> &stack)
{
	if(stack.empty())
		return;

	int topEle = stack.top();
	stack.pop();
	sortStack(stack);
	insertInStack(stack,topEle);
	return ;
}
