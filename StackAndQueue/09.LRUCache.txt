class LRUCache {
public:


    struct Node
    {
        Node* nextPtr;
        Node* prevPtr;
        int val;
        int key;
        Node(int k,int v)
        {
            nextPtr = nullptr;
            prevPtr= nullptr;
            val=v;
            key=k;
        }
    };

        int len;
    Node* start;
    Node* end;
    map<int,Node*>m;

    LRUCache(int capacity) {

        start=new Node(0,0);
        end = new Node(0,0);
        start->nextPtr=end;
        end->prevPtr=start;
        len=capacity;
        m.clear();
    }

    Node* removeNode(Node* node)
    {
        Node* prevNode=node->prevPtr;
        Node* nextNode = node->nextPtr;
        prevNode->nextPtr = nextNode;
        nextNode->prevPtr = prevNode;
        node->prevPtr = nullptr;
        node->nextPtr = nullptr;
        return node;
    }

    void putNodeInFront(Node* node)
    {
        Node* temp=start->nextPtr;
        start->nextPtr = node;
        node->prevPtr=start;
        node->nextPtr = temp;
        temp->prevPtr=node;

    }

    int get(int key) {
        if(m.find(key)==m.end())
            return -1;
        else
        {
            Node* nodeRemoved = removeNode(m[key]);
            putNodeInFront(nodeRemoved);
            return nodeRemoved->val;
        }
    }



    void put(int key, int value)
    {
        // ✅ If key already exists, update value and move to front
        if (m.find(key) != m.end()) {
            Node* existing = m[key];
            existing->val = value;
            removeNode(existing);
            putNodeInFront(existing);
            return;
        }

        // ✅ Key does not exist
        if (m.size() == len) {
            // Remove LRU node from list and map
            Node* lru = end->prevPtr;
            removeNode(lru);
            m.erase(lru->key);
            delete lru;
        }

        // Insert new node
        Node* node = new Node(key, value);
        putNodeInFront(node);
        m[key] = node;

    }
};

/**
 * Your LRUCache object will be instantiated and called as such:
 * LRUCache* obj = new LRUCache(capacity);
 * int param_1 = obj->get(key);
 * obj->put(key,value);
 */
