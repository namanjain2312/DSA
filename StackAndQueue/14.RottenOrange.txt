class Solution {
public:
    int orangesRotting(vector<vector<int>>& grid) {
        int countGood = 0;
        int out=0,n,a,b;
        queue<pair<int,int>>q;
        vector<vector<int>> vis(grid.size(), vector<int>(grid[0].size(), 0));


        for(int i;i<grid.size();i++)
        {
            for(int j=0;j<grid[i].size();j++)
            {
                if(grid[i][j]==1)
                    countGood++;
                else if(grid[i][j]==2)
                {
                    q.push({i,j});
                    vis[i][j]=1;
                }
            }
        }
        while(!q.empty())
        {
            if(countGood==0)
                return out;
            n=q.size();
            for(int i=0;i<n;i++)
            {
                a=q.front().first;
                b=q.front().second;
                if(a-1>=0 && vis[a-1][b]==0)
                {
                    vis[a-1][b]=1;
                    if(grid[a-1][b]==1)
                    {
                        countGood--;
                        q.push({a-1,b});
                    }
                }
                if(b-1>=0 && vis[a][b-1]==0)
                {
                    vis[a][b-1]=1;
                    if(grid[a][b-1]==1)
                    {
                        countGood--;
                        q.push({a,b-1});
                    }
                }
                if(a+1 < grid.size() && vis[a+1][b]==0)
                {
                    vis[a+1][b]=1;
                    if(grid[a+1][b]==1)
                    {
                        countGood--;
                        q.push({a+1,b});
                    }
                }
                if(b+1 < grid[0].size() && vis[a][b+1]==0)
                {
                    vis[a][b+1]=1;
                    if(grid[a][b+1]==1)
                    {
                        countGood--;
                        q.push({a,b+1});
                    }
                }
                q.pop();
            }
            out++;
        }
        if(countGood==0)
                return out;
        return -1;
    }
};
