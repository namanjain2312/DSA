12.SlidingWindowMaximum

//Last Modified On : 02 March 2025
//Could be done via Dequeue

class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        priority_queue<pair<int,int>>pq;
        int start=0,end=0;
        int n=nums.size();
        vector<int>out;

        for(int i=0;i<n;i++)
        {
            while(!pq.empty() && nums[i]>=pq.top().first)
                pq.pop();
            pq.push({nums[i],i});

            if(i+1>=k)
            {
                while(!pq.empty() && pq.top().second<(i-k+1))
                    pq.pop();
                out.push_back(pq.top().first);
            }

        }
        return out;





    }
};
