15.StockSpanProblem

class StockSpanner {
public:
    stack<pair<int,int>>s;

    StockSpanner() {
        while(!s.empty())
            s.pop();
    }

    int next(int price) {
        int temp=1;

        while(true)
        {
            if(s.empty() || s.top().first>price)
            {
                s.push({price,temp});
                break;
            }
            else
            {
                temp=s.top().second+temp;
                s.pop();
            }
        }
        return temp;
    }
};
