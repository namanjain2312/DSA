11.LargestRectangleInAHistogram

class Solution {
public:
    int largestRectangleArea(vector<int>& heights)
    {
        int n=heights.size(),temp,out=0;
        vector<int> greaterLines(n,-1);
        stack<pair<int,int>>s;


        for(int i=0;i<n;i++)
        {
            temp=1;
            while(true)
            {
                if(s.empty())
                {
                    s.push({heights[i],temp});
                    greaterLines[i] += temp;
                    break;
                }
                else if(s.top().first<heights[i])
                {
                    s.push({heights[i],temp});
                    greaterLines[i] += temp;
                    break;
                }
                else
                {
                    temp +=s.top().second;
                    s.pop();
                }
            }
        }

        while(!s.empty())
            s.pop();

        for(int i=n-1;i>=0;i--)
        {
            temp=1;
            while(true)
            {
                if(s.empty())
                {
                    s.push({heights[i],temp});
                    greaterLines[i] += temp;
                    break;
                }
                else if(s.top().first<heights[i])
                {
                    s.push({heights[i],temp});
                    greaterLines[i] += temp;
                    break;
                }
                else
                {
                    temp +=s.top().second;
                    s.pop();
                }
            }
        }

        for(int i=0;i<n;i++)
            out=std::max(out,heights[i]*greaterLines[i]);
        return out;

    }
};
