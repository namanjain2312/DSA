2.DFS

class Solution {
  public:
    // Function to return a list containing the DFS traversal of the graph.

    void dfs(vector<vector<int>>& adj,vector<int>&vis,vector<int>&out,int index)
    {
        if(vis[index]==1)
            return ;
        else
            vis[index]=1;

        out.push_back(index);

        int pointer;
        for(int i=0;i<adj[index].size();i++)
        {
            pointer = adj[index][i];
            dfs(adj,vis,out,pointer);
        }
    }

    vector<int> dfsOfGraph(vector<vector<int>>& adj) {
        int n = adj.size();
        vector<int>vis (n,0);
        vector<int>out;
        dfs(adj,vis,out,0);
        return out;



    }
};
