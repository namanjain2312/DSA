11.BipartiteCheckUsingBFS
QuestionLink - https://leetcode.com/problems/is-graph-bipartite/description/

class Solution {
public:

    bool isBipartite(vector<vector<int>>& graph)
    {
        int n=graph.size();
        vector<int>vis(n,-1);
        queue<pair<int,int>>q;
        int curr,color;

        for(int i=0;i<n;i++)
        {
            if(vis[i]==-1)
            {
                q.push({i,0});

                while(!q.empty())
                {
                    curr=q.front().first;
                    color=q.front().second;
                    q.pop();

                    if(vis[curr]%2==-1)
                    {
                        vis[curr]=color;
                        for(int j=0;j<graph[curr].size();j++)
                            q.push({graph[curr][j],color+1});
                    }
                    else if(vis[curr]%2==0 && color%2!=0)
                        return false;
                    else if(vis[curr]%2!=0 && color%2==0)
                        return false;
                }
            }
        }
        return true;

    }
};
