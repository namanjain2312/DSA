class Solution {
  public:
    // Function to return Breadth First Traversal of given graph.
    vector<int> bfsOfGraph(vector<vector<int>> &adj) {
        vector<int>out;
        queue<int>q;
        vector<int>vis(adj.size(),0);
        int index;
        q.push(0);

        while(!q.empty())
        {
            index=q.front();
            q.pop();
            if(vis[index]==1)
                continue;
            vis[index]=1;
            out.push_back(index);
            for(int i=0;i<adj[index].size();i++)
                q.push(adj[index][i]);

        }
        return out;

    }
};
