10.NumberOfIslands
QuestionLink - https://leetcode.com/problems/number-of-islands/description/


class Solution {
public:

    void dfs(vector<vector<char>>&grid,vector<vector<int>>&vis,int i,int j)
    {
        if(vis[i][j]==1)
            return ;
        vis[i][j]=1;

        if(i-1>=0 && grid[i-1][j]=='1')
            dfs(grid,vis,i-1,j);
        if(j-1>=0 && grid[i][j-1]=='1')
            dfs(grid,vis,i,j-1);
        if(i+1<grid.size() && grid[i+1][j]=='1')
            dfs(grid,vis,i+1,j);
        if(j+1<grid[0].size() && grid[i][j+1]=='1')
            dfs(grid,vis,i,j+1);

        return ;
    }

    int numIslands(vector<vector<char>>& grid) {
        int n=grid.size();
        int m = grid[0].size();
        int count=0;
        vector<vector<int>>vis(n,vector<int>(m,0));

        for(int i=0;i<n;i++)
        {
            for(int j=0;j<m;j++)
            {
                if(vis[i][j]==0 && grid[i][j]=='1')
                {
                    dfs(grid,vis,i,j);
                    count++;
                }
            }
        }

        return count;

    }
};
