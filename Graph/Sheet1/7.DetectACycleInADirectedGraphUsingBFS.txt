7.Detect A Cyle in A directed graph using BFS
QuestionLink - https://leetcode.com/problems/course-schedule/description/
QuestionLink - https://www.geeksforgeeks.org/problems/detect-cycle-in-a-directed-graph/1

Steps
1>We will use Kahn's Algo for topo sort
2>Generate Topo Sort for the graph
3>If Topo sort if of size N then no cycle else if size<N then cycle was there
4>Optimization -> instead of storing element in out vector in topo sort , we could simply maintain a
count variable and if count==N then return false , else return true;

SC : O(N)
TC : O(V+E)


class Solution {
  public:
    bool isCyclic(int V, vector<vector<int>> &edges) {
        queue<int>bfs;
        vector<int>indg(V,0);
        int count=0;
        vector<vector<int>>adj(V,vector<int>());


        for(int i=0;i<edges.size();i++)
        {
            adj[edges[i][0]].push_back(edges[i][1]);
            indg[edges[i][1]]++;
        }


        for(int i=0;i<V;i++)
        {
            if(indg[i]==0)
                bfs.push(i);
        }

        while(!bfs.empty())
        {
            int curr=bfs.front();
            count++;
            bfs.pop();

            for(int i=0;i<adj[curr].size();i++)
            {
                indg[adj[curr][i]]--;
                if(indg[adj[curr][i]]==0)
                    bfs.push(adj[curr][i]);
            }
        }
        if(count==V)
            return false;
        else
            return true;
    }
};
