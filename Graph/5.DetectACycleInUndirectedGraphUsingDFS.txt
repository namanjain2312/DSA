class Solution {
  public:
    bool checkCycleDfs(int src,int parent,vector<vector<int>> &adj,vector<int>&vis)
    {
        if(vis[src]==1)
            return true;
        vis[src]=1;
        int curr;
        for(int i=0;i<adj[src].size();i++)
        {
            curr=adj[src][i];
            if(parent!=curr)
            {
                if(checkCycleDfs(curr,src,adj,vis)==true)
                    return true;
            }
        }
        return false;
    }
    bool isCycle(vector<vector<int>>& adj) {

        vector<int> vis(adj.size(),0);

        for(int i=0;i<adj.size();i++)
        {
            if(vis[i]==0)
            {
                if(checkCycleDfs(i,-1,adj,vis)==true)
                    return true;
            }
        }
        return false;

    }
};
