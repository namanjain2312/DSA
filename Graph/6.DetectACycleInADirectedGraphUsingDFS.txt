6.DetectACycleInADirectedGraphUsingDFS

1>Previous approach of marking vis = 1 for all those element when it came in dfs will not work here
because consider a graph in that
1->2->5
3->4->5

Now 5 could be visited by 2 and 4 but still our graph do not have cycle in it , so that the issue here
so for the same what are the steps we follow

Steps
1>Do dfs of a node1(Parent)
2>Mark it as vis[node1]=1
2>Do dfs of all the node whoose parent is the node1
3>Mark all nodes as soon as there dfs is called
4>After completion of dfs of all child nodes of node 1
5>Mark vis[node1]=0;
6>Also if in between step 2 and step6 we encounter node1 again then we have a cycle
7>in this apporach we need to make two vis array one is vis and another one is dfsVis

TimeComplexity : O(N+E)
SpaceComplexity : O(N) + O(N)
Auxcilary Stack Space : O(N)

bool detectCycle(vector<int>&vis,vector<int>&dfsVis,int curr,vector<vector<int>>&adj)
{
    if(vis[curr]==1)
        return false;

    vis[curr]=1;
    dfsVis[curr]=1;

    for(int i=0;i<adj[curr].size();i++)
    {
        if(dfsVis[adj[curr][i]]==1)
            return true;
        else if(detectCycle(vis,dfsVis,adj[curr][i],adj)==true)
            return true;
    }


    dfsVis[curr]=0;
    return false;

}

class Solution {
  public:
    bool isCyclic(int V, vector<vector<int>> &edges) {
       vector<vector<int>>adj(V,vector<int>());
       vector<int>vis(V,0);
       vector<int>dfsVis(V,0);

       for(int i=0;i<edges.size();i++)
            adj[edges[i][0]].push_back(edges[i][1]);

        for(int i=0;i<V;i++)
        {
            if(vis[i]==0)
            {
                if(detectCycle(vis,dfsVis,i,adj)==true)
                    return true;

            }
        }
        return false;
    }
};
