13.0or1Matrix
Problem Link - https://leetcode.com/problems/01-matrix/description/

Simple level order bfs , put all zero in a node and do level order traversal


class Solution {
public:
    vector<vector<int>> updateMatrix(vector<vector<int>>& mat)
    {
        int n=mat.size();
        int m=mat[0].size();
        vector<vector<int>>vis(n,vector<int>(m,-1));
        queue<pair<int,int>>q;
        int count=0;
        int level,row,col;


        for(int i=0;i<n;i++)
        {
            for(int j=0;j<m;j++)
            {
                if(mat[i][j]==0)
                {
                    q.push({i,j});
                }
            }
        }

        while(!q.empty())
        {
            level=q.size();
            while(level--)
            {
                row=q.front().first;
                col=q.front().second;
                q.pop();
                if(vis[row][col]!=-1)
                    continue;
                vis[row][col]=count;
                if(row+1<n && vis[row+1][col]==-1)
                    q.push({row+1,col});
                if(row-1>=0 && vis[row-1][col]==-1)
                    q.push({row-1,col});
                if(col+1<m && vis[row][col+1]==-1)
                    q.push({row,col+1});
                if(col-1>=0 && vis[row][col-1]==-1)
                    q.push({row,col-1});
            }
            count++;
        }

        return vis;


    }
};
