7.NumberOfProvinces
Question Link - https://leetcode.com/problems/number-of-provinces/description/

Simple concepts of number fo connected components used


class Solution {
public:

    void dfs(int src,vector<vector<int>>& isConnected,vector<int>&vis)
    {
        vis[src]=1;
        for(int j=0;j<isConnected.size();j++)
        {
            if(vis[j]==0 && isConnected[src][j]==1)
                dfs(j,isConnected,vis);
        }
    }

    int findCircleNum(vector<vector<int>>& isConnected)
    {
        int n=isConnected.size();
        int count=0;
        vector<int>vis(n,0);

        for(int i=0;i<n;i++)
        {
            if(vis[i]==0)
            {
                count++;
                dfs(i,isConnected,vis);
            }
        }
        return count;

    }
};
