14.SurroundedRegions.txt

Problem Link -  https://leetcode.com/problems/surrounded-regions/description/
Dfs all the zeroes which are on the edges and mark them with vis[i][j]=1 , now rest all can be marked as 'X'

class Solution {
public:

    void dfs(vector<vector<int>>&vis,vector<vector<char>>& board,int row,int col)
    {
        vis[row][col]=1;

        if(row+1<board.size() && board[row+1][col]=='O' && vis[row+1][col]==0)
            dfs(vis,board,row+1,col);
        if(col+1<board[0].size() && board[row][col+1]=='O' && vis[row][col+1]==0)
            dfs(vis,board,row,col+1);
        if(row-1>=0 && board[row-1][col]=='O' && vis[row-1][col]==0)
            dfs(vis,board,row-1,col);
        if(col-1>=0 && board[row][col-1]=='O' && vis[row][col-1]==0)
            dfs(vis,board,row,col-1);

        return ;
    }

    void solve(vector<vector<char>>& board)
    {
        int n,m;
        n=board.size();
        m=board[0].size();

        vector<vector<int>>vis(n,vector<int>(m,0));

        for(int i=0;i<n;i++)
        {
            if(vis[i][0]==0 && board[i][0]=='O')
                dfs(vis,board,i,0);
            if(vis[i][m-1]==0 && board[i][m-1]=='O')
                dfs(vis,board,i,m-1);
        }

        for(int j=0;j<m;j++)
        {
            if(vis[0][j]==0 && board[0][j]=='O')
                dfs(vis,board,0,j);
            if(vis[n-1][j]==0 && board[n-1][j]=='O')
                dfs(vis,board,n-1,j);
        }

        for(int i=0;i<n;i++)
        {
            for(int j=0;j<m;j++)
            {
                if(vis[i][j]==0)
                    board[i][j]='X';
            }
        }

        return ;




    }
};
