class Solution {
public:

    bool findAns(vector<vector<int>>&grid,int ans)
    {

    }

    int swimInWater(vector<vector<int>>& grid) {
        int low,high;
        int n=grid.size();
        int low=INT_MAX;
        int high=INT_MIN;


        for(int i=0;i<n;i++)
        {
            for(int j=0;j<n;j++)
            {
                low=min(grid[i][j],low);
                high=max(grid[i][j],high);
            }
        }


    }
};
