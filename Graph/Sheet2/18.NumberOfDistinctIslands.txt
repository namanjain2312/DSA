18.NumberOfDistinctIslands

Problem Link -  https://www.geeksforgeeks.org/problems/number-of-distinct-islands/1

Subtract each node from its intial/starting node values and and store the answer in a set and return sthat


// User function Template for C++

class Solution {
  public:

    vector<int>dir={-1,1};

    void dfs(int row,int col,int a,int b,set<vector<pair<int,int>>>&s,
    vector<vector<int>>&vis,vector<vector<int>>&grid,
    vector<pair<int,int>>&curr)
    {
        vis[row][col]=1;
        curr.push_back({row-a,col-b});

        for(int i=0;i<2;i++)
        {
            int c,r;
            r=row+dir[i];
            c=col;
            if(r>=0 && r<grid.size() && vis[r][c]==-1 &&grid[r][c]==1)
                dfs(r,c,a,b,s,vis,grid,curr);
            r=row;
            c=col+dir[i];

            if(c>=0 && c<grid[0].size() && vis[r][c]==-1 && grid[r][c]==1)
                dfs(r,c,a,b,s,vis,grid,curr);
        }

        return ;



    }


    int countDistinctIslands(vector<vector<int>>& grid) {
        set<vector<pair<int,int>>>s;
        int n=grid.size();
        int m=grid[0].size();

        vector<vector<int>>vis(n,vector<int>(m,-1));

        for(int i=0;i<n;i++)
        {
            for(int j=0;j<m;j++)
            {
                if(grid[i][j]==1 && vis[i][j]==-1)
                {
                    vector<pair<int,int>>curr;
                    dfs(i,j,i,j,s,vis,grid,curr);
                    s.insert(curr);
                }

            }
        }
        return s.size();
    }
};
