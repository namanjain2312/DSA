26.FindEventualSafeStates
Problem Link - https://leetcode.com/problems/find-eventual-safe-states/description/

Solution -  Reverse the edges of the graph and the do the topo sort and return all the node in that topo sort

class Solution {
public:
    vector<int> eventualSafeNodes(vector<vector<int>>& graph)
    {
        queue<int>q;
        vector<int>out;
        int n=graph.size();
        vector<int>inDegree(n,0);
        vector<vector<int>>adj(n,vector<int>());

        for(int i=0;i<n;i++)
        {
            for(int j=0;j<graph[i].size();j++)
                adj[graph[i][j]].push_back(i);
        }

        for(int i=0;i<n;i++)
        {
            for(int j=0;j<adj[i].size();j++)
                inDegree[adj[i][j]]++;
        }

        for(int i=0;i<n;i++)
        {
            if(inDegree[i]==0)
                q.push(i);
        }

        while(!q.empty())
        {
            int curr=q.front();
            q.pop();
            out.push_back(curr);

            for(int i=0;i<adj[curr].size();i++)
            {
                inDegree[adj[curr][i]]--;
                if(inDegree[adj[curr][i]]==0)
                    q.push(adj[curr][i]);
            }
        }
        sort(out.begin(),out.end());
        return out;



    }
};
