16.Word Ladder

Problem Link -

We construct a graph where edges exist between words that differ by one character.
Starting from endWord, we perform BFS to find the shortest path to beginWord,
incrementing the level at each breadth. The first time we find a valid transformation
to beginWord, we return the path length.


// In this question we will tranverse in oppostie order i.e from distination
//to source in order to get the shortest path

class Solution {
public:

    bool isDiffValid(string s1,string s2)
    {
        int count=0;
        int i=0;
        int j=0;
        while(i<s1.size() && j<s2.size())
        {
            if(s1[i]!=s2[j])
            {
                count++;
                if(count>1)
                    return false;
            }
            i++;
            j++;
        }
        return true;
    }


    int ladderLength(string beginWord, string endWord, vector<string>& wordList)
    {
        if(endWord.size()!=beginWord.size())
            return 0;

        int n=wordList.size();
        unordered_map<string,vector<string>>adj;
        unordered_map<string,int>vis;


        int out=0;

        for(int i=0;i<n;i++)
        {
            vis[wordList[i]]=-1;
            for(int j=i+1;j<n;j++)
            {
                if(isDiffValid(wordList[i],wordList[j]))
                {
                    adj[wordList[i]].push_back(wordList[j]);
                    adj[wordList[j]].push_back(wordList[i]);
                }
            }
        }

        queue<string>q;
        q.push(endWord);
        if(vis[endWord]!=-1)
            return 0;

        while(!q.empty())
        {
            int level=q.size();
            out++;

            for(int i=0;i<level;i++)
            {
                string node=q.front();

                vis[node]=1;
                q.pop();
                if(isDiffValid(node,beginWord))
                    return out+1;
                for(int j=0;j<adj[node].size();j++)
                {
                    if(vis[adj[node][j]]==-1)
                        q.push(adj[node][j]);
                }
            }
        }
        return 0;

    }
};
