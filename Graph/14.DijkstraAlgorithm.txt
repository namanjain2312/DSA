3>Works only for positive weights and undirected graph and no negative cycle graph
4>Now my Solution have two codes , above one is to find the shortest path and print the path
5>Other is to find the shortest path value from a source to all the vertex
6>Why we cannot use Queue instead of PQ , ans is to avoid unnecessary iteration as in queue we will not go in
the most optimized order and hence repeating the path again and again unecesaaryliy, so while using PQ
we will greedily pick up the path but in Queue we will pick all path and unecessary long path and then short path

class Solution {
  public:

    vector<int> shortestPath(int n, int m, vector<vector<int>>& edges)
    {
        //dist , node , parentNode
        priority_queue<vector<int>,vector<vector<int>>,greater<vector<int>>>pq;
        pq.push({0,1,-1});
        //dist , parent
        vector<pair<int,int>>vis(n+1,{-1,-1});

        vector<vector<pair<int,int>>> adj(n+1);

        for(int i=0;i<edges.size();i++)
        {
            adj[edges[i][0]].push_back({edges[i][1],edges[i][2]});
            adj[edges[i][1]].push_back({edges[i][0],edges[i][2]});
        }

        while(!pq.empty())
        {
            int dist=pq.top()[0];
            int currNode=pq.top()[1];
            int parentNode=pq.top()[2];
            pq.pop();
            if(vis[currNode].first!=-1 && vis[currNode].first<=dist)
            {
                continue;
            }
            vis[currNode]={dist,parentNode};

            for(int i=0;i<adj[currNode].size();i++)
            {
                pq.push({dist+adj[currNode][i].second,adj[currNode][i].first,currNode});
            }

        }

        vector<int>out;
        if(vis[n].first==-1)
        {
            out.push_back(-1);
            return out;
        }
        int temp=n;

        while(temp!=-1)
        {
            out.push_back(temp);
            temp=vis[temp].second;
        }

        out.push_back(vis[n].first);
        reverse(out.begin(),out.end());
        return out;

    }
};

class Solution {
  public:
    struct comp {
    bool operator()(const pair<int,int>& p1, const pair<int,int>& p2) {
        return p1.first > p2.first;
    }
};

    vector<int> dijkstra(vector<vector<pair<int, int>>> &adj, int src)
    {
        priority_queue<pair<int,int>,vector<pair<int,int>>,comp> pq;
        int n=adj.size();
        int curr,dist;

        vector<int>vis(n,-1);
        pq.push({0,src});

        while(!pq.empty())
        {
            dist=pq.top().first;
            curr=pq.top().second;
            if(vis[curr]!=-1 && vis[curr]<=dist)
            {
                pq.pop();
                continue;
            }

            vis[curr]=dist;

            for(int i=0;i<adj[curr].size();i++)
            {
                pq.push({dist+adj[curr][i].second,adj[curr][i].first});
            }
            pq.pop();
        }

        return vis;

    }
};
