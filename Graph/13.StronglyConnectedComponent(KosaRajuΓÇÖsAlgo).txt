class Solution {
  public:

    void dfs(int curr,stack<int>&s,vector<int>&vis,vector<vector<int>> &adj)
    {
        if(vis[curr]==1)
            return ;

        vis[curr]=1;

        for(int i=0;i<adj[curr].size();i++)
            dfs(adj[curr][i],s,vis,adj);

        s.push(curr);

    }

    int dfs2(int curr,vector<int>&vis,vector<vector<int>> &transpose)
    {
        if(vis[curr]==1)
            return 0;


        vis[curr]=1;

        for(int i=0;i<transpose[curr].size();i++)
            dfs2(transpose[curr][i],vis,transpose);

        return 1;
    }

    int kosaraju(vector<vector<int>> &adj) {
        int n=adj.size();
        int out=0;
        stack<int> s;
        vector<vector<int>> transpose(n,vector<int>());
        vector<int>vis(n,0);

        for(int i=0;i<n;i++)
            dfs(i,s,vis,adj);

        for(int i=0;i<n;i++)
        {
            for(int j=0;j<adj[i].size();j++)
            {
                transpose[adj[i][j]].push_back(i);
            }
        }



        for(int i=0;i<n;i++)
            vis[i]=0;

        while(!s.empty())
        {
            out=out+dfs2(s.top(),vis,transpose);
            s.pop();
        }
        return out;
    }
};
