vector<int> Solution::dNums(vector<int> &A, int B)
{
    vector<int>out;
    int j=0;
    if(B>A.size())
        return out;
    unordered_map<int,int>m;
    for(int i=0;i<B;i++)
        m[A[i]]++;
    out.push_back(m.size());
    for(int i=B;i<A.size();i++)
    {
        m[A[i]]++;
        if(m[A[j]]==1)
            m.erase(A[j]);
        else
            m[A[j]]--;
        out.push_back(m.size());
        j++;
    }
    return out;
}
