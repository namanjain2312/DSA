//We have done it using dfs instead of Bfs , observe that

class Solution {
 public:

     void dfs(vector<vector<int>>& photo,int row,int col,int color,int prevColor)
     {
         photo[row][col]=color;

         if(row+1<photo.size() && photo[row+1][col]==prevColor)
             dfs(photo,row+1,col,color,prevColor);
         if(row-1>=0 && photo[row-1][col]==prevColor)
             dfs(photo,row-1,col,color,prevColor);
         if(col+1<photo[0].size()&&photo[row][col+1]==prevColor)
             dfs(photo,row,col+1,color,prevColor);
         if(col-1>=0&&photo[row][col-1]==prevColor)
             dfs(photo,row,col-1,color,prevColor);
         return ;
     }


     vector<vector<int>> floodFill(vector<vector<int>>& image, int sr, int sc, int color)
     {
         vector<vector<int>> photo;
         photo=image;
         if(photo[sr][sc]!=color)
         {
             dfs(photo,sr,sc,color,photo[sr][sc]);
         }
         return photo;
     }
 };
