class KthLargest {
public:
    priority_queue<int,vector<int>,greater<int>> pq;
    int requestedLargest;

    KthLargest(int k, vector<int>& nums) {
        requestedLargest = k;
        while(!pq.empty())
            pq.pop();

        for(int i=0;i<nums.size();i++)
            pq.push(nums[i]);

        while(pq.size()>k)
            pq.pop();

    }

    int add(int val) {
        pq.push(val);
        while(pq.size()>requestedLargest)
            pq.pop();
        return pq.top();

    }
};
