class Codec {
public:
    // Encodes a tree to a single string.
    string serialize(TreeNode* root) {
        string out;
        TreeNode *tempNode;
        queue<TreeNode*>q;
        q.push(root);

        while(!q.empty())
        {
            tempNode = q.front();
            q.pop();

            if(tempNode==nullptr)
                out=out+"#,";
            else
            {
                out=out+to_string(tempNode->val) + ",";
                q.push(tempNode->left);
                q.push(tempNode->right);
            }
        }
        out.pop_back();
        return out;
    }

    // Decodes your encoded data to tree.
    TreeNode* deserialize(string data) {
        if(data=="#")
            return nullptr;
        int i=0;
        queue<TreeNode*>q;
        TreeNode *dummy,*tempNode;
        vector<string>v;
        string temp;

        for(int i=0;i<data.size();i++)
        {
            if(data[i]==',')
            {
                v.push_back(temp);
                temp="";
            }
            else
                temp.push_back(data[i]);
        }

        dummy=new TreeNode(0);
        tempNode = new TreeNode(stoi(v[0]));
        dummy->left = tempNode;
        q.push(tempNode);
        i=1;
        while(i<v.size())
        {
            tempNode=q.front();
            q.pop();

            if(v[i]!="#")
            {
                TreeNode *leftNode =new TreeNode(stoi(v[i]));
                tempNode->left=leftNode;
                q.push(leftNode);
            }
            i++;
            if(i==v.size())
                break;
            if(v[i]!="#")
            {
                TreeNode *rightNode = new TreeNode(stoi(v[i]));
                tempNode->right= rightNode;
                q.push(rightNode);
            }
            i++;
        }
        return dummy->left;
    }
};
