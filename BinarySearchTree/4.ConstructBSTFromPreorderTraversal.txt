class Solution {
public:
    TreeNode* buildTree(int start,int end,vector<int>&preorder)
    {
        if(start>end)
            return nullptr;
        TreeNode *root=new TreeNode(preorder[start]);
        int seperationIndex=-1;
        for(int i=start+1;i<=end;i++)
        {
            if(preorder[i]>preorder[start])
            {
                seperationIndex = i;
                break;
            }
        }
        if(seperationIndex==-1)
            root->left=buildTree(start+1,end,preorder);
        else
            root->left=buildTree(start+1,seperationIndex-1,preorder);

        if(seperationIndex!=-1)
            root->right=buildTree(seperationIndex,end,preorder);
        else
            root->right=nullptr;

        return root;
    }

    TreeNode* bstFromPreorder(vector<int>& preorder) {
        return buildTree(0,preorder.size()-1,preorder);
    }
};
