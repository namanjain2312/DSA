7.AllocateMinimumNumberOfPages

bool canBooksBeDistributed(int maxPage,vector<int>&A,int B)
{
    int currSum=0,studentCount=1;
    for(int i=0;i<A.size();i++)
    {
        if (A[i] > maxPage)
            return false;

        if(currSum+A[i]>maxPage)
        {
            studentCount++;
            currSum = A[i];
            if (studentCount > B)
                return false;
        }
        else
        {
            currSum = A[i] + currSum;
        }
    }

   return true;
}
int Solution::books(vector<int> &A, int B) {
    if (B > A.size()) return -1;

    int low=INT_MIN,high=0,mid,result;
    for(int i=0;i<A.size();i++)
    {
        low=max(low,A[i]);
        high = high+A[i];
    }

    while(high>=low)
    {
        mid=(high+low)/2;
        if(canBooksBeDistributed(mid,A,B)==true)
        {
            result = mid;
            high=mid-1;

        }
        else
        {
            low=mid+1;
        }

    }
    return result;
}
