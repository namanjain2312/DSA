class Solution {
public:
    int singleNonDuplicate(vector<int>& nums) {
        int low,high,mid;
        low=0;
        high=nums.size()-1;
        while(low<=high)
        {
            mid=(low+high)/2;
            if(mid+1<nums.size() && nums[mid]==nums[mid+1] && (high-mid+1)%2==0)
                high=mid-1;
            else if(mid-1>=0 && nums[mid-1]==nums[mid] && (mid-low+1)%2==0)
                low=mid+1;
            else if(mid+1<nums.size() && nums[mid]==nums[mid+1] && (high-mid+1)%2!=0)
                low=mid+2;
            else if(mid-1>=0 && nums[mid-1]==nums[mid] && (mid-low+1)%2!=0)
                high=mid-2;
            else if(mid-1>=0 && mid+1<nums.size() && nums[mid]!=nums[mid-1] && nums[mid]!=nums[mid+1])
                return nums[mid];
            else if(mid-1>=0 && mid+1==nums.size() && nums[mid-1]!=nums[mid])
                return nums[mid];
            else if(mid+1<nums.size() && mid-1<0 && nums[mid+1]!=nums[mid])
                return nums[mid];
            else if(mid-1<0 && mid+1==nums.size())
                return nums[mid];
        }
        return -1;
    }
};
