class Solution {
public:

    double calMedian(vector<int>&nums1,vector<int>&nums2)
    {
        int low,high,mid1,n1=nums1.size(),n2=nums2.size(),n,half,l1,l2,r1,r2,mid2;
        low=0;
        high=n1;
        n=n1+n2;
        half=(n+1)/2;
        while(low<=high)
        {
            l1=INT_MIN;
            l2=INT_MIN;
            r1=INT_MAX;
            r2=INT_MAX;

            mid1=(low+high)/2;
            mid2=half-mid1;


            if(mid1>0)
                l1=nums1[mid1-1];
            if(mid1<n1)
                r1=nums1[mid1];
            if(mid2>0)
                l2=nums2[mid2-1];
            if(mid2<n2)
                r2=nums2[mid2];

            if(max(l1,l2)<=min(r1,r2))
            {
                if(n%2==0)
                    return (double)((max(l1,l2)+min(r1,r2))/2.00);
                else
                    return (double) (max(l1,l2));
            }
            else if(l1>r2)
                high=mid1-1;
            else
                low=mid1+1;
        }
        return 0.00;

    }

    double findMedianSortedArrays(vector<int>& nums1, vector<int>& nums2) {
        if(nums1.size()>nums2.size())
            return calMedian(nums2,nums1);
        else
            return calMedian(nums1,nums2);
    }
};
