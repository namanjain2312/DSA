class Solution {
  public:

    int findKElement(vector<int>&a,vector<int>&b,int k)
    {
        int l1,l2,r1,r2,mid1,mid2,low,high,n1,n2,n;
        n1=a.size();
        n2=b.size();
        n=n1+n2;
        low = max(0, k - n2);
        high = min(k, n1);

        while(low<=high)
        {
            mid1=(low+high)/2;
            mid1=min(k,mid1);
            mid2=k-mid1;
            //cout<<mid1<<" "<<mid2<<"\n";
            l1=INT_MIN;
            l2=INT_MIN;
            r1=INT_MAX;
            r2=INT_MAX;

            if(mid1>0)
                l1=a[mid1-1];
            if(mid2>0)
                l2=b[mid2-1];
            if(mid1<n1)
                r1=a[mid1];
            if(mid2<n2)
                r2=b[mid2];

            if(max(l1,l2)<=min(r1,r2))
            {
                return max(l1,l2);
            }
            else if(l1>r2)
                high=mid1-1;
            else
                low=mid1+1;
        }
        return -1;

    }
    int kthElement(vector<int>& a, vector<int>& b, int k) {
        if(a.size()>b.size())
            return findKElement(b,a,k);
        else
            return findKElement(a,b,k);
    }
};
