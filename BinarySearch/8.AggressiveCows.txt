#include <iostream>
#include <bits/stdc++.h>
using namespace std;

bool isArrangementPossible(int dist,vector<int> &stalls,int c)
{
    int lastCowPlaced=0;
    int cowPlacedCount=1;
    for(int i=1;i<stalls.size();i++)
    {
        if(stalls[i]-stalls[lastCowPlaced]>=dist)
        {
            lastCowPlaced = i;
            cowPlacedCount++;
            if(cowPlacedCount==c)
                return true;
        }
    }
    return false;
}

int main() {
    int t;
    cin>>t;
    while(t--)
    {
        int n,c,a,low,high,mid;
        vector<int>stalls;
        cin>>n>>c;
        for(int i=0;i<n;i++)
        {
            cin>>a;
            stalls.push_back(a);
        }
        sort(stalls.begin(),stalls.end());
        low=stalls[0];
        high=stalls[n-1];

        while(high>=low)
        {
            mid=(high+low)/2;
            if(isArrangementPossible(mid,stalls,c))
                low=mid+1;
            else
                high=mid-1;

        }
        cout<<high<<"\n";
    }
}
