struct TrieNode
{
    vector<TrieNode*> itr;
    char val;//character
    int countOfWordsEqualTo=0;//count of word equal to this word
    int countOfWordsStartingWith=0;//count of words having this prefix

    TrieNode(char val)
    {
        this->val=val;
        this->countOfWordsEqualTo=0;
        this->countOfWordsStartingWith=0;
        for(int i=0;i<26;i++)
            this->itr.push_back(nullptr);
    }
};

class Trie
{
    TrieNode *head=nullptr;
    public :
        Trie()
        {
            clear(head);
            head=new TrieNode('a');
        }

        //Insert a word in trie
        void insert(string word)
        {
            TrieNode *curr=head;
            int i=0;

            while(i<word.size())
            {
                //Case when no trie node is there
                if(curr->itr[word[i]-'a']==nullptr)
                    curr->itr[word[i]-'a']=new TrieNode(word[i]);

                //Case when the word end
                if(i==word.size()-1)
                    curr->itr[word[i]-'a']->countOfWordsEqualTo++;

                //Case for counting prefix
                curr->itr[word[i]-'a']->countOfWordsStartingWith++;

                //Traversal
                curr=curr->itr[word[i]-'a'];

                i++;
            }
        }

        bool checkCompleteness(string word)
        {
            TrieNode *curr=head;
            int i=0;

            while(i<word.size())
            {
                if(curr->itr[word[i]-'a']->countOfWordsEqualTo==0)
                    return false;

                //Traversal
                curr=curr->itr[word[i]-'a'];
                i++;
            }
            return true;
        }

        //Delete and clear whole Trie
        void clear(TrieNode* node) {
            if (node==nullptr)
                return;
            for (int i=0;i<26;i++)
                clear(node->itr[i]);
            delete node;
        }
};

string completeString(int n, vector<string> &a){
    Trie trie = Trie();
    string ans="None";


    for(int i=0;i<a.size();i++)
        trie.insert(a[i]);

    for(int i=0;i<a.size();i++)
    {
        if(trie.checkCompleteness(a[i]))
        {
            if(ans=="None")
                ans=a[i];
            else if(ans.size()<a[i].size())
                ans=a[i];
            else if(ans.size()==a[i].size())
                ans=min(a[i],ans);
        }
    }
    return ans;




}
