4.NumberOfDistinctSubstringsInString

Link - https://www.geeksforgeeks.org/problems/count-of-distinct-substrings/1

#include <bits/stdc++.h>

struct TrieNode
{
    TrieNode* itr[26];
    char val;//character

    TrieNode(char val)
    {
        this->val=val;
        for(int i=0;i<26;i++)
            this->itr[i]=nullptr;
    }
};



// Helper function to recursively delete all nodes
void clearTrie(TrieNode* node) {
    if (node == nullptr)
        return;
    for (int i = 0; i < 26; ++i)
        clearTrie(node->itr[i]);
    delete node;
}


int countDistinctSubstring(string s) {
    TrieNode *head=nullptr;
    head=new TrieNode('a');

    int ans=0;

    for(int i=0;i<s.size();i++)
    {
        TrieNode* node=head;

        for(int j=i;j<s.size();j++)
        {
            if(node->itr[s[j]-'a']==nullptr)
            {
                node->itr[s[j]-'a'] = new TrieNode(s[j]);
                ans++;
            }
            node=node->itr[s[j]-'a'];
        }
    }

    clearTrie(head); // clean up memory
    return ans+1;

}
