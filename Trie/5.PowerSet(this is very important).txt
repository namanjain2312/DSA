class Solution{
	public:
	    vector<string>out;

	    void powerSet(string &s,int index,string &curr)
	    {
	        if(index==s.size())
	        {
	            out.push_back(curr);
	            return;
	        }

	        powerSet(s,index+1,curr);
	        curr.push_back(s[index]);
	        powerSet(s,index+1,curr);
	        curr.pop_back();
	        return ;
	    }

		vector<string> AllPossibleStrings(string s){
		    string curr;
		    powerSet(s,0,curr);
		    sort(out.begin(),out.end());
		    return out;
		}
};
