class Solution {
public:
    Node* copyRandomList(Node* head) {
        unordered_map<Node*,Node*>m;
        Node* temp = head;
        m[nullptr]=nullptr;


        while(temp!=nullptr)
        {
            Node* dummy = new Node(temp->val);
            m[temp]=dummy;
            temp=temp->next;
        }

        temp=head;
        while(temp!=nullptr)
        {
            m[temp]->next=m[temp->next];
            m[temp]->random=m[temp->random];
            temp=temp->next;
        }


        return m[head];

    }
};
