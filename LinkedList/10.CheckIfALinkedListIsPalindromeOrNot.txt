//Try again in this Space complexity is O(n) , but this could be done in Space O(1)
//Reach middle , reverse the second part of list and then traverse from start and //second part of link list and then compare
public:
    bool isPalindrome(ListNode* head) {
        stack<int>s;
        ListNode *slow=head,*fast=head;

        while(fast!=nullptr)
        {
            if(fast->next==nullptr)
            {
                slow=slow->next;
                break;
            }
            s.push(slow->val);
            slow=slow->next;
            fast=fast->next->next;
        }
        while(slow!=nullptr)
        {
            if(s.top()==slow->val)
            {
                s.pop();
                slow=slow->next;
            }
            else
                return false;
        }
        return true;
    }
