class Solution {
public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        ListNode *head=nullptr,*pre=nullptr,*dummy=nullptr;
        dummy = new ListNode();
        pre = new ListNode();
        dummy = pre;


        int carry=0,sum=0;


        while(l1!=nullptr || l2 !=nullptr || carry!=0)
        {
            sum=0;
            if(l1!=nullptr)
            {
                sum+=l1->val;
                l1=l1->next;
            }
            if(l2!=nullptr)
            {
                sum+=l2->val;
                l2=l2->next;
            }
            sum+=carry;

            carry = sum/10;
            head = new ListNode(sum % 10);
            pre->next=head;
            pre=head;
        }
        return dummy->next;

    }
};
