vector<int> MinimumCoins(int n)
{
    vector<int>v={1, 2, 5, 10, 20, 50, 100, 500, 1000};
    int count=0;
    int temp;
    vector<int>out;
    int j=v.size()-1;
    while(n>0)
    {
        if(n>=v[j])
        {
            temp=n/v[j];
            n=n-temp*v[j];
            while(temp--)
                out.push_back(v[j]);
        }
        else
            j--;
    }
    return out;
}
