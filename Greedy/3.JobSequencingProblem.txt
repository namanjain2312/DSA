class Solution {
  public:


    vector<int> jobSequencing(vector<int> &deadline, vector<int> &profit)
    {
        set<int>s;
        vector<pair<int,int>>pq;
        int n=deadline.size();
        int out=0;
        int maxDeadline =0;
        int count=0;

        for(int i=0;i<n;i++)
        {
            pq.push_back({profit[i],deadline[i]});
            maxDeadline=max(deadline[i],maxDeadline);
        }
        sort(pq.begin(),pq.end());

        for(int i=1;i<=maxDeadline;i++)
            s.insert(i);
        int j=pq.size()-1;

        while(j>=0)
        {
            if(s.size()==0)
                break;
            auto i = upper_bound(s.begin(),s.end(),pq[j].second);
            if(i==s.begin())
                j--;
            else
            {
                i--;
                out+=pq[j].first;
                j--;
                s.erase(i);
                count++;
            }

        }
        return {count,out};

    }
};
