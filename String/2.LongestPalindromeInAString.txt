class Solution {
public:
    string longestPalindrome(string s)
    {
        int n=s.size();
        vector<vector<int>>dp(n,vector<int>(n,0));
        int windowSize=2;
        int i,j;
        pair<int,int>p;
        p={0,0};

        for(int i=0;i<n;i++)
        {
            dp[i][i]=1;
            if(i+1<n && s[i]==s[i+1])
            {
                p.first=i;
                p.second=i+1;
                dp[i][i+1]=1;
            }
        }

        while(windowSize<n)
        {
            i=0;
            while(i+windowSize<n)
            {
                j=i+windowSize;
                if(s[i]==s[j] && dp[i+1][j-1]==1)
                {
                    p.first=i;
                    p.second=j;
                    dp[i][j]=1;

                }
                i++;
            }
            windowSize++;
        }

        return s.substr(p.first,p.second-p.first+1);

    }
};
