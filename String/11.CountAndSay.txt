11. CountAndSay

class Solution {
public:
    string funct(string ans)
    {
        string out="";
        int prev;
        int count;
        int i=1;
        prev=ans[0]-'0';
        count=1;

        while(i<ans.size())
        {
            if(ans[i]-'0'==prev)
                count++;
            else
            {
                out.append(to_string(count));
                out.append(to_string(prev));
                prev=ans[i]-'0';
                count=1;
            }
            i++;
        }
        out.append(to_string(count));
        out.append(to_string(prev));
        return out;
    }

    string countAndSay(int n)
    {
        string ans="1";

        for(int i=2;i<=n;i++)
        {
            ans=funct(ans);
        }
        return ans;
    }
};
