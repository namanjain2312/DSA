10.CheckForAnagrams

Approach 1
class Solution {
public:
    bool isAnagram(string s, string t) {
        if(s.size()!=t.size())
            return false;

        map<char,int>m;

        for(int i=0;i<s.size();i++)
        {
            m[s[i]]++;
            m[t[i]]--;
        }

        for(auto i = m.begin(); i!=m.end(); i++)
        {
            if(i->second!=0)
                return false;
        }
        return true;
    }
};

Approch 2
class Solution {
public:
    bool isAnagram(string s, string t) {
        sort(s.begin(),s.end());
        sort(t.begin(),t.end());
        return s==t;
    }
};
