12. Compare Version Numbers

class Solution {
public:
    int compareVersion(string version1, string version2) {
        vector<int>a,b;
        int i=0;
        string temp="";

        while(i<version1.size())
        {
            if(version1[i]!='.')
                temp.push_back(version1[i]);
            else
            {
                a.push_back(stoi(temp));
                temp="";
            }
            i++;
        }
        if(temp!="")
            a.push_back(stoi(temp));

        temp="";
        i=0;

        while(i<version2.size())
        {
            if(version2[i]!='.')
                temp.push_back(version2[i]);
            else
            {
                b.push_back(stoi(temp));
                temp="";
            }
            i++;

        }
        if(temp!="")
            b.push_back(stoi(temp));

        while(a.size()!=b.size())
        {
            if(a.size()>b.size())
                b.push_back(0);
            else
                a.push_back(0);
        }

        for(int i=0;i<a.size();i++)
        {
            if(a[i]>b[i])
                return 1;
            else if(a[i]<b[i])
                return -1;
        }

        return 0;




    }
};
