class Solution {
public:

    int computeLPS(string &pattern,string mainString) {
        string temp=pattern+'#'+mainString;
    vector<int> lps(temp.size(),0 ); // lps[0] is always 0
    int len = 0;          // length of the previous longest prefix suffix
    int i = 1;            // start from the second character

    while (i < temp.size()) {
        if (temp[i] == temp[len]) {
            len++;
            lps[i] = len;
            i++;
        } else {
            if (len != 0) {
                // Instead of starting over, use the previous LPS value
                len = lps[len - 1];
            } else {
                // No match found, so set lps[i] to 0
                lps[i] = 0;
                i++;
            }
        }
    }
    for(int i=pattern.size()+1;i<temp.size();i++)
        {
            if(lps[i]==pattern.size())
                return 1;
        }
        return 0;
}

    int repeatedStringMatch(string a, string b) {
        int k=b.size()/a.size();
        if(b.size()%a.size()!=0)
            k++;

        string b1="";
        for(int i=0;i<k;i++)
        {
            b1+=a;
        }
        if(computeLPS(b,b1)==1)
        {
            return k;
        }
        else if(computeLPS(b,b1+a)==1)
        {
            return k+1;
        }
        return -1;
    }
};
