1.ReverseWordsInAString

class Solution {
public:
    string reverseWords(string s)
    {
        string out="";
        int j=s.size()-1;
        string temp="";

        while(j>=0)
        {
            if(s[j]==' ' && temp=="")
                ;
            else if(s[j]==' ' && temp!="")
            {
                reverse(temp.begin(),temp.end());
                out+=temp+" ";
                temp="";
            }
            else
                temp.push_back(s[j]);
            j--;
        }

        if(temp!="")
        {
            reverse(temp.begin(),temp.end());
            out+=temp+" ";
            temp="";
        }
        out.pop_back();

        return out;

    }
};
