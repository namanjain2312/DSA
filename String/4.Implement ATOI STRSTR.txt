4.Implement ATOI STRSTR

class Solution {
public:
    int myAtoi(string s)
    {
        int i=0;
        int n=s.size();
        int temp=1;
        string integer;

        while(i<n && s[i]==' ')
            i++;

        if(i==n)
            return 0;
        if(s[i]=='+')
            i++;
        else if(s[i]=='-')
        {
            i++;
            temp=-1;
        }

        while(i<n && s[i]=='0')
            i++;

        if(i==n)
            return 0;

        while(i<n && s[i]-'0'>=0 && s[i]-'0'<=9)
        {
            integer.push_back(s[i]);
            i++;
        }
        if(integer.size()==0)
            return 0;

        if(integer.size()>11)
        {
            if(temp==-1)
                return INT_MIN;
            else
                return INT_MAX;
        }
        long long int ans=0;
        i=0;
        cout<<"Integer is "<<integer<<"\n";

        while(i<integer.size())
        {
            ans+= (integer[i]-'0')*pow(10,integer.size()-i-1);
            i++;
        }

        ans=ans*temp;
        if(ans<INT_MIN)
            return INT_MIN;
        else if(ans>INT_MAX)
            return INT_MAX;
        else
            return (int)ans;



    }
};
