class Solution {
public:
    int romanToInt(string s)
    {
        int out=0;
        map<string,int>m{{"I",1},{"V",5},{"X",10},{"L",50},{"C",100},{"D",500},{"M",1000}
        , {"IV",4},{"IX",9},{"XL",40},{"XC",90},{"CD",400},{"CM",900}};

        for(int i=0;i<s.size();i++)
        {
            if(i+1<s.size() && m.find(s.substr(i,2))!=m.end())
            {
                out+=m[s.substr(i,2)];
                i++;
            }
            else
                out+=m[s.substr(i,1)];

        }

        return out;




    }
};
