6.LeftViewOfBinaryTree

// Last Attempted On 26 Feb 2025
// Normal DFS will not work here as DFS will not consider the case let say root have
//two child left and right but the left child have no children but right have a child
//A , answer would root,left child and A but DFS will only provide the answer as
//root , left child
//But we will still do it with DFS as the code for it would be smaller
//But for now i am doing with BFS
class Solution {
  public:

    vector<int> leftView(Node *root) {

        Node* tempNode = root;
        vector<int>output;
        queue<Node*> bfs;
        bfs.push(tempNode);
        int levelSize = 0;

        while(!bfs.empty())
        {
            tempNode = bfs.front();
            if(tempNode==nullptr)
            {
                bfs.pop();
                continue;
            }
            output.push_back(tempNode->data);
            levelSize = bfs.size();
            while(levelSize--)
            {
                tempNode = bfs.front();
                if(tempNode->left!=nullptr)
                    bfs.push(tempNode->left);
                if(tempNode->right!=nullptr)
                    bfs.push(tempNode->right);
                bfs.pop();
            }
        }
        return output;

    }
};
