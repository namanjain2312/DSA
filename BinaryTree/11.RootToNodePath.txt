class Solution {
  public:

   void leafNodePaths(Node* node,vector<vector<int>> &output,vector<int> &currentPath)
   {
        if(node==nullptr)
            return ;

        currentPath.push_back(node->data);

        if(node->left==nullptr && node->right==nullptr)
            output.push_back(currentPath);
        else
        {
            leafNodePaths(node->left,output,currentPath);
            leafNodePaths(node->right,output,currentPath);
        }
        currentPath.pop_back();
        return ;
   }

    vector<vector<int>> Paths(Node* root) {
        vector<vector<int>> output;
        vector<int> currentPath ;
        Node* node = root;
        leafNodePaths(root,output,currentPath);

        return output;


    }
};
