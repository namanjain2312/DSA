// Tree Node
struct Node
{
    int data;
    Node* left;
    Node* right;
};

// Utility function to create a new Tree Node
Node* newNode(int val)
{
    Node* temp = new Node;
    temp->data = val;
    temp->left = NULL;
    temp->right = NULL;

    return temp;
}


vector <int> bottomView(Node *root);

// Function to Build Tree
Node* buildTree(string str)
{
    // Corner Case
    if(str.length() == 0 || str[0] == 'N')
        return NULL;

    // Creating vector of strings from input
    // string after spliting by space
    vector<string> ip;

    istringstream iss(str);
    for(string str; iss >> str; )
        ip.push_back(str);

    // Create the root of the tree
    Node* root = newNode(stoi(ip[0]));

    // Push the root to the queue
    queue<Node*> queue;
    queue.push(root);

    // Starting from the second element
    int i = 1;
    while(!queue.empty() && i < ip.size()) {

        // Get and remove the front of the queue
        Node* currNode = queue.front();
        queue.pop();

        // Get the current node's value from the string
        string currVal = ip[i];

        // If the left child is not null
        if(currVal != "N") {

            // Create the left child for the current node
            currNode->left = newNode(stoi(currVal));

            // Push it to the queue
            queue.push(currNode->left);
        }

        // For the right child
        i++;
        if(i >= ip.size())
            break;
        currVal = ip[i];

        // If the right child is not null
        if(currVal != "N") {

            // Create the right child for the current node
            currNode->right = newNode(stoi(currVal));

            // Push it to the queue
            queue.push(currNode->right);
        }
        i++;
    }

    return root;
}

//Function to return a list containing the bottom view of the given tree.

//It consists of two parameter , horizontal dist from root (centre node) ,
//Depth of each node
//For every horizontal dist we would have one answer and we would select the
//depthmost answer in it ;

class Solution {
  public:

    void bottomTraversal(Node* node,map<int,pair<int,Node*>>&horizontalDistantNode,int depth,int breadth)
    {
        if(node==nullptr)
            return ;
        if(horizontalDistantNode.find(breadth)!=horizontalDistantNode.end())
        {
            if(horizontalDistantNode[breadth].first <= depth)
                horizontalDistantNode[breadth] = {depth,node};
        }
        else
        {
            horizontalDistantNode.insert({breadth,{depth,node}});
        }
        bottomTraversal(node->left,horizontalDistantNode,depth+1,breadth-1);
        bottomTraversal(node->right,horizontalDistantNode,depth+1,breadth+1);
    }

    vector <int> bottomView(Node *root)
    {
        vector<int>output;
        Node* node = root;
        map<int,pair<int,Node*>>horizontalDistantNode;
        int depth=0;
        int breadth=0;
        bottomTraversal(node,horizontalDistantNode,0,0);
        for(auto i : horizontalDistantNode)
        {
            output.push_back(i.second.second->data);
        }

        return output;
    }
};
