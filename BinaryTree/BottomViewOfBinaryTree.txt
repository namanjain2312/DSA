//Function to return a list containing the bottom view of the given tree.
//// Last Attempted On 26 Feb 2025
//It consists of two parameter , horizontal dist from root (centre node) ,
//Depth of each node
//For every horizontal dist we would have one answer and we would select the
//depthmost answer in it ;

class Solution {
  public:

    void bottomTraversal(Node* node,map<int,pair<int,Node*>>&horizontalDistantNode,int depth,int breadth)
    {
        if(node==nullptr)
            return ;
        if(horizontalDistantNode.find(breadth)!=horizontalDistantNode.end())
        {
            if(horizontalDistantNode[breadth].first <= depth)
                horizontalDistantNode[breadth] = {depth,node};
        }
        else
        {
            horizontalDistantNode.insert({breadth,{depth,node}});
        }
        bottomTraversal(node->left,horizontalDistantNode,depth+1,breadth-1);
        bottomTraversal(node->right,horizontalDistantNode,depth+1,breadth+1);
    }

    vector <int> bottomView(Node *root)
    {
        vector<int>output;
        Node* node = root;
        map<int,pair<int,Node*>>horizontalDistantNode;
        int depth=0;
        int breadth=0;
        bottomTraversal(node,horizontalDistantNode,0,0);
        for(auto i : horizontalDistantNode)
        {
            output.push_back(i.second.second->data);
        }

        return output;
    }
};
