17.LCAInBinaryTree

class Solution {
public:

    TreeNode* dfs(TreeNode* tempNode, TreeNode* p, TreeNode* q)
    {
        if(tempNode==nullptr)
            return nullptr;
        if(tempNode==p||tempNode==q)
            return tempNode;

        TreeNode *left,*right;
        left = dfs(tempNode->left,p,q);
        right= dfs(tempNode->right,p,q);


        if(left!=nullptr&&right!=nullptr)
            return tempNode;

        if(left!=nullptr)
            return left;
        else
            return right;
    }

    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        TreeNode *tempNode = root;
        return dfs(tempNode,p,q);

    }
};
