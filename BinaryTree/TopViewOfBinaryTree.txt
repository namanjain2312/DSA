class Solution {
  public:
    // Function to return a list of nodes visible from the top view
    // from left to right in Binary Tree.
    void topViewTraversal(Node* node,map<int,pair<int,Node*>>&breadthWiseNode,int breadth,int depth)
    {
        if(node==nullptr)
            return ;
        if(breadthWiseNode.find(breadth)!=breadthWiseNode.end())
        {
            if(breadthWiseNode[breadth].first>depth)
                breadthWiseNode[breadth]={depth,node};
        }
        else
            breadthWiseNode.insert({breadth,{depth,node}});
        topViewTraversal(node->left,breadthWiseNode,breadth-1,depth+1);
        topViewTraversal(node->right,breadthWiseNode,breadth+1,depth+1);
    }

    vector<int> topView(Node *root) {
        vector<int>output;
        Node* node=root;
        map<int,pair<int,Node*>>breadthWiseNode;
        int breadth=0;
        int depth=0;
        topViewTraversal(node,breadthWiseNode,0,0);

        for(auto i : breadthWiseNode)
            output.push_back(i.second.second->data);

        return output;

    }
};
