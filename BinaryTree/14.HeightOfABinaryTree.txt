class Solution {
public:
    int maxDepth(TreeNode* root) {
        int maxDepth = 0,levelSize=0;
        queue<TreeNode*>bfs;
        TreeNode* node = root;
        if(node==nullptr)
            return maxDepth;
        bfs.push(node);

        while(!bfs.empty())
        {
            maxDepth++;
            levelSize = bfs.size();
            for(int i=0;i<levelSize;i++)
            {
                node=bfs.front();
                if(node->left!=nullptr)
                    bfs.push(node->left);
                if(node->right!=nullptr)
                    bfs.push(node->right);
                bfs.pop();
            }
        }
        return maxDepth;

    }
};
