class Solution {
public:
    int postIndex;

    TreeNode* build(vector<int>&inorder,vector<int>&postorder,int inStartIndex,int inEndIndex,unordered_map<int,int> &m)
    {
        if(inStartIndex>inEndIndex)
            return nullptr;
        postIndex--;
        TreeNode *root=new TreeNode();

        root->val= postorder[postIndex];
        int index = m[postorder[postIndex]];
        root->right=build(inorder,postorder,index+1,inEndIndex,m);
        root->left = build(inorder,postorder,inStartIndex,index-1,m);
        return root;
    }

    TreeNode* buildTree(vector<int>& inorder, vector<int>& postorder) {
        postIndex = postorder.size();
        unordered_map<int,int> m;
        for(int i=0;i<postorder.size();i++)
            m[inorder[i]]=i;
        return build(inorder,postorder,0,postorder.size()-1,m);
    }
};
