class Solution {
public:
    int uniquePaths(int m, int n) {
        long long totalDist = n+m-2;
        long long horizontalDist = min(n-1,m-1);
        long long temp=1ll;
        long long out=1ll;

        while(temp<=horizontalDist)
        {
            out=(out*totalDist)/temp;
            temp++;
            totalDist--;
        }

        return out;
    }
};
