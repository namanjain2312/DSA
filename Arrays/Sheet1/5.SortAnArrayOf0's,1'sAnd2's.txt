5.SortAnArrayOf0's,1'sAnd2's

class Solution {
public:
    void sortColors(vector<int>& nums) {
        int start=0;
        int mid=0;
        int n=nums.size();
        int end=n-1;

        while(mid <= end) {

            if(nums[mid] == 0) {
                swap(nums[start], nums[mid]);
                start++;
                mid++;
            }
            else if(nums[mid] == 1) {
                mid++;
            }
            else
            {
                swap(nums[mid], nums[end]);
                end--;
            }
        }

        return ;

    }
};
