22.LargestSubarrayWithKSum

class Solution {
  public:
    int maxLength(vector<int>& arr) {
        map<int,int>m;
        int sum=0;
        int out=0;


        for(int i=0;i<arr.size();i++)
        {
            sum+=arr[i];
            if(sum==0)
                out=max(i+1,out);
            if(m.find(sum)==m.end())
                m[sum]=i;

            out=max(out,i-m[sum]);

        }
        return out;

    }
};
