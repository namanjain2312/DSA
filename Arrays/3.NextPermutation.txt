class Solution {
public:

    void nextPermutation(vector<int>& nums)
    {
        int breakPoint=-1,n=nums.size(),temp;
        temp=nums[n-1];

        for(int i=n-2;i>=0;i--)
        {
            if(nums[i]<nums[i+1])
            {
                breakPoint = i;
                break;

            }
        }
        if(breakPoint==-1)
        {
            reverse(nums.begin(), nums.end());
            return ;
        }
        else
        {
            temp=INT_MAX;

            for(int i=breakPoint+1;i<n;i++)
            {
                if(nums[i]>nums[breakPoint] && temp>nums[i])
                    temp=nums[i];
            }

            for(int i=n-1;i>=breakPoint+1;i--)
            {
                if(nums[i]==temp)
                {
                    temp=nums[breakPoint];
                    nums[breakPoint]=nums[i];
                    nums[i]=temp;
                    break;
                }
            }

            reverse(nums.begin() + breakPoint + 1, nums.end());

            return;
        }
    }
};
