class Solution {
public:
    double pow(double a,long long b)
    {
        double out=1;
        if(b==1)
            return a;
        else if(b==0)
            return 1;
        else if(b%2==0)
        {
            return out*pow(a*a,b/2);
        }
        else
        {
            return out*a*pow(a*a,(b-1)/2);
        }
    }

    double myPow(double x, int n) {

        double ans=1;

        if(n<0)
        {
            return ans/pow(x,-1ll*n);
        }
        else
            return pow(x,n);

    }
};
