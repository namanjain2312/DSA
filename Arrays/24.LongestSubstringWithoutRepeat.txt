//It has Time Complexity as 1*N
class Solution {
public:

    int lengthOfLongestSubstring(string s)
    {
        map<char,int>m;
        int left=-1,right=0,count=0;
        for(int i=0;i<s.size();i++)
        {
            right=i;
            if(m.find(s[i])==m.end())
            {
                m[s[i]]=i;
                count=max(count,right-left);
            }
            else
            {
                left=max(left,m[s[i]]);
                m[s[i]]=i;
                count=max(count,right-left);
            }
        }
        return count;
    }
};


//It has Time Complexity as 2*N
class Solution {
public:
    int lengthOfLongestSubstring(string s)
    {
        map<char,int>m;
        int count=0;
        int lastPreIndex=-1,currIndex;

        for(int i=0;i<s.size();i++)
        {
            // cout<<"i is "<<i;

            if(m.find(s[i])==m.end())
            {
                // cout<<" Add "<<s[i];
                m[s[i]]=i;
                count=max(count,(int)m.size());
            }
            else
            {
                currIndex=m[s[i]];
                for(int j=lastPreIndex+1;j<=currIndex;j++)
                {
                    // cout<<" Erase "<<s[j]<<" ";
                    m.erase(s[j]);
                }
                m[s[i]]=i;
                // cout<<" Add "<<s[i];
                count=max(count,(int)m.size());
                lastPreIndex=currIndex;
            }
            // cout<<"\n";
        }
        return count;
    }
};
