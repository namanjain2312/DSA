// Basic Approach : is to use a max heap , insert all the elements in the max heap
// Then pop k-1 element from it , now return the topmost element of the heap
// TC : O(n)*log(n) + O(k)*log(n)
// SC : O(n)
// Second Approach : would be always maitain heap size k and only maintain k elements ,
// using min heap
// Third approach : Use quick select algoritm
// TC : O(n) average , O(n^2) worse
// SC : O(1)

//Better
class Solution {
public:
    int findKthLargest(vector<int>& nums, int k) {
        priority_queue<int,vector<int>,greater<int>>pq;

        for(int i=0;i<nums.size();i++)
        {
            if(pq.size()<k)
                pq.push(nums[i]);
            else if(pq.top()<nums[i])
            {
                pq.pop();
                pq.push(nums[i]);
            }
        }
        return pq.top();
    }
};

class Solution {

public:
    int findKthLargest(vector<int>& nums, int k) {
        priority_queue<int>pq;
        int n;
        n=nums.size();
        for(int i=0;i<n;i++)
            pq.push(nums[i]);

        while(--k)
        {
            pq.pop();
        }
        return pq.top();
    }
};
